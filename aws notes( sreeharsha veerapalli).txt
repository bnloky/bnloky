Bucket policy : its is way to give a access to the s3 bucket for different users.
 # aws--->aws policy generator ( we can generate the policy for the s3,sqs,sns,vpc endpoint policy, and IAM policy)

   "Version": "2012-10-17",
    "Statement": [{
        "Sid": "AllowGetObject",

            "AWS": "*"   ----------------------------------------> '*' wildcard means that anonyums accesss  which anyone in the world can access.
        },
        "Effect": "Allow",
        "Action": "s3:GetObject",

        "Condition": {
            "StringEquals": {
                "aws:PrincipalOrgID": ["o-aa111bb222"]
            }
        }
    }
   ]
  }
